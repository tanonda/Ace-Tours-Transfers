Drop this file into a React + TypeScript project (e.g. Replit configured for TSX) as src/PopsyDashboards.tsx

// Utilities
const fmtVT = (n?: number) => (n == null ? '-' : n.toLocaleString('en-US') + ' VT')
const pad = (v: number) => String(v).padStart(2, '0')
function isoDate(daysAgo = 0) {
  const d = new Date()
  d.setDate(d.getDate() - daysAgo)
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
}

// ------------------------------
// Timeboxed fetch helper with TypeScript generics
async function apiGet<T>(url: string, timeout = 8000): Promise<T> {
  const controller = new AbortController()
  const id = setTimeout(() => controller.abort(), timeout)
  try {
    const res = await fetch(url, { signal: controller.signal })
    clearTimeout(id)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
    const data = (await res.json()) as T
    return data
  } finally {
    clearTimeout(id)
  }
}

function generateMockSeries(days = 30) {

  const bookings: number[] = []

  for (let i = days - 1; i >= 0; i--) {
    labels.push(isoDate(i))

    bookings.push(Math.round(10 + Math.random() * 80))
    signups.push(Math.round(1 + Math.random() * 20))
  }
  return { labels, revenue, bookings, signups }
}

// Hooks (REST-style) — each returns { data, loading, error, refresh }
function useFetchWithFallback<T>(url: string, fallback: T, deps: any[] = []) {
  const [data, setData] = useState<T | null>(null)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  async function load() {
    setLoading(true)
    setError(null)
    try {
      const d = await apiGet<T>(url)
      setData(d)
    } catch (err: any) {
      console.warn('Fetch failed for', url, err?.message || err)
      setData(fallback)
      setError(err?.message || 'fallback')
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    load()
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps)

  return { data, loading, error, refresh: load }
}

// Admin hooks
function useAdminRevenueData(days = 30) {
  const url = `${API_BASE}/admin/analytics/revenue?days=${days}`
  // Expected response: { points: Array<{date:string,value:number}> }
  const fallback = { points: generateMockSeries(days).labels.map((d, i) => ({ date: d, value: generateMockSeries(days).revenue[i] })) }
  return useFetchWithFallback<{ points: RevenuePoint[] }>(url, fallback, [url])
}

function useAdminBookingsData(days = 30) {

  const fallback = { points: generateMockSeries(days).labels.map((d, i) => ({ date: d, value: generateMockSeries(days).bookings[i] })) }
  // map to BookingPoint on the consumer side
  return useFetchWithFallback<{ points: BookingPoint[] }>(url, fallback, [url])
}

}

const chartPalette = [THEME.accent4, THEME.accent1, THEME.accent3, THEME.accent2]

function makeLineData(labels: string[], data: number[], label = 'Series') {
  return {
    labels,
    datasets: [
      {
        label,
        data,
        fill: true,
        tension: 0.3,
        backgroundColor: 'rgba(77,150,255,0.12)',
        borderColor: THEME.accent4,
        pointRadius: 2
      }
    ]
  }
}

function makeBarData(labels: string[], data: number[], label = 'Value', color = THEME.accent4) {
  return {
    labels,
    datasets: [
      {
        label,
        data,
        backgroundColor: color,
        borderRadius: 6
      }
    ]
  }
}

function makePieData(labels: string[], data: number[]) {
  return {
    labels,
    datasets: [
      {
        data,
        backgroundColor: chartPalette,
        hoverOffset: 6
      }
    ]
  }
}

// Router & wrapper
export default function PopsyDashboards() {
  const [mode, setMode] = useState<'admin' | 'customer'>(() => (location.hash.startsWith('#/customer') ? 'customer' : 'admin'))
  const [route, setRoute] = useState<string>(() => location.hash || `#/admin/dashboard`)

  useEffect(() => {
    const onHash = () => {
      const h = location.hash || '#/admin/dashboard'
      setRoute(h)
      setMode(h.startsWith('#/customer') ? 'customer' : 'admin')
    }
    window.addEventListener('hashchange', onHash)
    onHash()
    return () => window.removeEventListener('hashchange', onHash)
  }, [])

  function navTo(h: string) {
    location.hash = h
  }

  const content = useMemo(() => {
    if (route.startsWith('#/admin/analytics')) return <AdminAnalytics />
    if (route.startsWith('#/customer/analytics')) return <CustomerAnalytics />
    if (mode === 'admin') return <AdminDashboard />
    return <CustomerDashboard />
  }, [route, mode])

  return (
    <div style={{ minHeight: '100vh', background: `linear-gradient(180deg, ${THEME.bg}, ${THEME.surface})`, color: THEME.text, fontFamily: 'Inter, ui-sans-serif, system-ui, -apple-system, "Segoe UI", Roboto, "Helvetica Neue", Arial' }}>

      <div style={{ display: 'flex', gap: 16, padding: 18 }}>
        <div style={{ width: 220 }}>
          <Sidebar mode={mode} setMode={m => { setMode(m); navTo(m === 'admin' ? '#/admin/dashboard' : '#/customer/dashboard') }} />

          <div style={{ display: 'flex', gap: 8, marginTop: 12 }}>
            <button className={`btn ${mode === 'admin' ? 'active' : ''}`} onClick={() => navTo('#/admin/dashboard')}>Admin</button>
            <button className={`btn ${mode === 'customer' ? 'active' : ''}`} onClick={() => navTo('#/customer/dashboard')}>Customer</button>
          </div>

          <div style={{ marginTop: 16, display: 'flex', flexDirection: 'column', gap: 8 }}>
            <button className="chip" onClick={() => navTo(mode === 'admin' ? '#/admin/analytics' : '#/customer/analytics')}>Open Analytics</button>
            <button className="chip" onClick={() => navTo(mode === 'admin' ? '#/admin/dashboard' : '#/customer/dashboard')}>Dashboard</button>
          </div>
        </div>

        <div style={{ flex: 1, display: 'flex', flexDirection: 'column' }}>
          {content}
        </div>
      </div>

      <style>{`
        :root{ --accent1: ${THEME.accent1}; --accent2:${THEME.accent2}; --accent3:${THEME.accent3}; --accent4:${THEME.accent4}; }
        .btn{ background: linear-gradient(90deg, var(--accent4), var(--accent1)); color:#031428; padding:8px 12px; border-radius:8px; border:none; font-weight:700; cursor:pointer }
        .btn.transparent{ background:transparent; color:rgba(230,238,243,0.9); border:1px solid rgba(255,255,255,0.04) }
        .btn.small{ padding:6px 8px; font-size:13px }
        .chip{ background:rgba(255,255,255,0.03); padding:8px 10px; border-radius:999; cursor:pointer } 
        .card{ background: linear-gradient(180deg, rgba(255,255,255,0.02), rgba(255,255,255,0.01)); color: ${THEME.text}; }
        table th, table td{ color: rgba(230,238,243,0.85); }
      `}</style>
    </div>
  )
}
